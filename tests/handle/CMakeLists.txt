add_executable(sim_card_probe probe.cpp)
set_target_properties(sim_card_probe PROPERTIES AUTOMOC ON)
target_link_libraries(sim_card_probe PRIVATE Lidar::SimCard)
target_compile_definitions(sim_card_probe PRIVATE QT_NO_KEYWORDS)
if(MSVC)
    target_compile_options(sim_card_probe PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(sim_card_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME sim_card_probe COMMAND sim_card_probe)
set_tests_properties(sim_card_probe PROPERTIES TIMEOUT 30)
if(WIN32)
    set_tests_properties(sim_card_probe PROPERTIES ENVIRONMENT_MODIFICATION
        "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
endif()
foreach(header IN ITEMS SimCardProtocol SimCardHandle)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${header}_alone.cpp" CONTENT "#include <${header}.h>\n")
    target_sources(sim_card_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${header}_alone.cpp")
endforeach()
